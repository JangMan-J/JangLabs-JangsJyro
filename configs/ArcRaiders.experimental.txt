# ── T.3/L.3 — ZR analog split (click-layer approximation) ────────────────────
# VDF: RT fires add_layer(t=0ms) → LMB(t=50ms) → remove_layer(t=150ms)
# JSM: split into soft-pull zone (silent) and full-pull zone (LMB)
# Gyro freeze is NOT implemented; see KNOWN LIMITATIONS above.
ZR_MODE = NO_SKIP                # enable independent soft-pull (ZR) and full-pull (ZRF)
ZR = NONE                        # soft-pull ZR = no output (replaces safe-tier LMOUSE)
ZRF = LMOUSE                     # full-pull ZR = LMB (fire on hard click)

# ── L.1 — Menu modeshift (held "-" = Select) ─────────────────────────────────
# VDF: Menu preset (Preset_1000006) swaps gyro ratchet, dpad, and face buttons.
# JSM: held "-" approximates a subset of Menu preset bindings.
# Note: JSM names Switch's "-" / "+" buttons as the literal symbols `-` / `+`,
# not the words MINUS / PLUS (operators.cpp:41-50).
# B.8 Menu D-pad (scroll nav at 10 Hz):
-,UP = SCROLLUP+
-,DOWN = SCROLLDOWN+
-,RIGHT = TAB
-,LEFT = ESC
# B.11 Menu system buttons:
-,- = E                          # Select-in-menu context = E
-,+ = Q                          # Menu-in-menu context = Q
# B.12 Menu face A/B (global HOLD_PRESS_TIME = 200 ms; matches VDF):
-,S = MMOUSE LCONTROL            # A in menu: tap = MiddleClick, hold = LCtrl  # approx
-,E = ESC LALT                   # B in menu: tap = Esc, hold = LAlt            # approx
